"""Fresh 1W measurement and independent effect oracle; no product check in oracle."""
import copy,hashlib,json,os,platform,statistics,subprocess,sys,time
from pathlib import Path
HOME=Path(__file__).resolve().parent
ROOT=HOME/'fixture'
COL='docs/investigations'
IX=COL+'/investigation-index.json'
ENV=dict(os.environ,NODE_OPTIONS='')
NODE=subprocess.check_output(['which','node'],text=True).strip()
def sha(raw): return hashlib.sha256(raw).hexdigest()
def git(*args): return subprocess.check_output(['git','-C',str(ROOT),*args])
def write(relative,data):
    dest=ROOT/relative
    dest.parent.mkdir(exist_ok=True,parents=True)
    dest.write_bytes(data if isinstance(data,bytes) else data.encode())
def doc(number): return COL+f'/perf-{number:05}.md'
def ident(number): return f'261001-perf-{number:05}'
def revision(number,text):
    digest=hashlib.sha256(b'investigation-index-entry-v2\0')
    for field in (ident(number),f'perf-{number:05}.md',text.replace('\r\n','\n').replace('\r','\n')):
        data=field.encode(); digest.update(str(len(data)).encode()+b':'+data+b'\0')
    return 'sha256:'+digest.hexdigest()
def members():
    # Raw native Git member tuples preserve path, mode, stage and OID (including deletions).
    return git('ls-files','--stage','-z').split(b'\0')
def without_index(rows): return [row for row in rows if not row.endswith(b'\t'+IX.encode())]
def workspace():
    return {str(p.relative_to(ROOT)):{'sha256':sha(p.read_bytes()),'mode':p.stat().st_mode & 0o777} for p in ROOT.rglob('*') if '.git' not in p.parts and p.is_file()}
def pending_index(): return json.loads(git('show',':'+IX))
def publish(raw): write(IX,json.dumps(raw,ensure_ascii=False,indent=2)+'\n')
def update_workspace(number,marker,title=None):
    text=(ROOT/doc(number)).read_text()
    text=text.replace('fixturecontent ',f'fixturecontent {marker} ',1)
    if title: text=text.replace(f'title: "独立记录 {ident(number)}"',f'title: "{title}"')
    write(doc(number),text)
    raw=json.loads((ROOT/IX).read_bytes())
    if title: raw['entries'][ident(number)]['title']=title
    raw['sourceRevision']['entries'][ident(number)]=revision(number,text)
    publish(raw)
def pristine():
    # Only this owned fixture is restored. No workspace outside ROOT is touched.
    git('read-tree','--reset','-u','HEAD')
    for p in (ROOT/COL/'_resources').rglob('extra.txt'): p.unlink()
    write('outside.txt','protected pending\n');git('add','--','outside.txt');git('rm','-q','--','delete-me.txt')
def native_prepare():
    pristine();update_workspace(7,'selected-B')
    write(COL+f'/_resources/{ident(7)}/extra.txt','z'*95+'\n')
def projected(before,number):
    expected=copy.deepcopy(before); working=json.loads((ROOT/IX).read_bytes())
    expected['entries'][ident(number)]=working['entries'][ident(number)]
    expected['sourceRevision']['entries'][ident(number)]=working['sourceRevision']['entries'][ident(number)]
    return expected

def run(version,group,number,label):
    prior=pending_index(); expected=projected(prior,number)
    before_members=members(); before_workspace=workspace()
    argv=[NODE,str(HOME/version/'investigation-report/scripts/check-investigations.ts'),'--root',str(ROOT),'stage',ident(number),'--scope','index']
    prefix=HOME/label
    with open(str(prefix)+'.stdout','wb') as out,open(str(prefix)+'.stderr','wb') as err:
        start=time.perf_counter();pid=os.fork()
        if pid==0:
            os.dup2(out.fileno(),1);os.dup2(err.fileno(),2);os.execve(NODE,argv,ENV)
        _,status,usage=os.wait4(pid,0)
        wall=time.perf_counter()-start
    stdout=Path(str(prefix)+'.stdout').read_bytes();stderr=Path(str(prefix)+'.stderr').read_bytes()
    actual=pending_index()
    effects={'exitSuccess':os.waitstatus_to_exitcode(status)==0,'fullTargetJsonMatches':actual==expected,'nonIndexPendingMembersModesOidsPreserved':without_index(before_members)==without_index(members()),'allWorkspaceBytesAndModesUnchanged':before_workspace==workspace(),'outsidePendingRetained':git('show',':outside.txt')==b'protected pending\n','stagedDeletionRetained':b'delete-me.txt' not in git('ls-files','-z').split(b'\0'),'scopeIndexDeclared':b'(scope: index)' in stdout and b'Report Markdown and attached resources remain outside this operation.' in stdout}
    row={'group':group,'version':version,'label':label,'selectedId':ident(number),'argv':argv,'wallSeconds':wall,'cpuSeconds':usage.ru_utime+usage.ru_stime,'userSeconds':usage.ru_utime,'systemSeconds':usage.ru_stime,'maxRssKiB':usage.ru_maxrss,'exitCode':os.waitstatus_to_exitcode(status),'stdoutSha256':sha(stdout),'stderrSha256':sha(stderr),'expectedTargetJsonSha256':sha(json.dumps(expected,sort_keys=True,ensure_ascii=False,separators=(',',':')).encode()),'actualTargetJsonSha256':sha(json.dumps(actual,sort_keys=True,ensure_ascii=False,separators=(',',':')).encode()),'pendingTree':git('write-tree').decode().strip(),'effects':effects,'effectsPassed':all(effects.values()),'NODE_OPTIONS':''}
    with (HOME/'observations.jsonl').open('a') as out: out.write(json.dumps(row,ensure_ascii=False)+'\n')
    print(json.dumps(row,ensure_ascii=False),flush=True)
    assert row['effectsPassed'],row
    return row

def native():
    native_prepare()
    for pair in range(1,4):
        for version in ('baseline','final'):
            git('reset','-q','HEAD','--',IX)
            run(version,'native-basic',7,f'native-pair{pair}-{version}')
def continuous(version):
    pristine()
    # Real P differs from H: staged A report/resource, pending publication and B refreshes.
    update_workspace(1,'first-A','连续A第一版')
    git('add','--',doc(1))
    write(COL+f'/_resources/{ident(1)}/extra.txt','A-owned staged member\n')
    git('add','--',COL+f'/_resources/{ident(1)}/extra.txt')
    run(version,'continuous-effects',1,f'continuous-{version}-A1')
    update_workspace(7,'first-B','连续B第一版')
    git('add','--',doc(7))
    run(version,'continuous-effects',7,f'continuous-{version}-B1')
    update_workspace(1,'refresh-A','连续A第二版')
    run(version,'continuous-effects',1,f'continuous-{version}-A2')
    update_workspace(7,'refresh-B','连续B第二版')
    run(version,'continuous-effects',7,f'continuous-{version}-B2')

def summary():
    rows=[json.loads(line) for line in (HOME/'observations.jsonl').read_text().splitlines()]
    groups={}
    for group in sorted({r['group'] for r in rows}):
        groups[group]={}
        for version in sorted({r['version'] for r in rows if r['group']==group}):
            selection=[r for r in rows if r['group']==group and r['version']==version]
            groups[group][version]={'count':len(selection),'wallMedianSeconds':statistics.median(r['wallSeconds'] for r in selection),'cpuMedianSeconds':statistics.median(r['cpuSeconds'] for r in selection),'rssMedianMiB':statistics.median(r['maxRssKiB']/1024 for r in selection),'allEffectsPassed':all(r['effectsPassed'] for r in selection)}
    (HOME/'summary.json').write_text(json.dumps({'groups':groups,'allEffectsPassed':all(r['effectsPassed'] for r in rows)},ensure_ascii=False,indent=2)+'\n')
    streams={}
    for row in rows:
        for kind in ('stdout','stderr'):
            raw=(HOME/(row['label']+'.'+kind)).read_bytes();streams.setdefault(sha(raw),raw.decode())
    (HOME/'streams.json').write_text(json.dumps(streams,ensure_ascii=False,indent=2)+'\n')
    print(json.dumps(groups,indent=2))

if __name__=='__main__':
    action=sys.argv[1]
    if action=='native':native()
    elif action=='continuous':
        for version in ('baseline','final'):continuous(version)
    elif action=='summary':summary()
    else:raise ValueError(action)

END FILE experiment.py
