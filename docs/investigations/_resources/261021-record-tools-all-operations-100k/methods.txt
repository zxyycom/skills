function classify(run, metrics, scenario) {
  if (run.timedOut) return "command-timeout";
  if (run.spawnError !== null) return "launch-failed";
  if (
    /JavaScript heap out of memory|Reached heap limit|Ineffective mark-compacts near heap limit/u.test(
      run.stderr.head
    )
  )
    return "node-heap-limit";
  if (metrics.value?.childSignal || run.signal) return "command-signalled";
  if (metrics.status !== "ok") return "measurement-unavailable";
  if (run.exitCode === scenario.expectedExit)
    return run.exitCode === 0 ? "expected-success" : "expected-rejection";
  return "unexpected-exit";

export async function prepareFixture(root, count) {

  const sources = [];
  for (let start = 0; start < count; start += 128) {
    const batch = [];
    for (let number = start; number < Math.min(count, start + 128); number++) {
      const source = formalSource(number);
      sources.push(source);
      batch.push(write(root, `${collection}/${source.sourcePath}`, source.text));

  assert.ok(gitDirectory.isDirectory() && !gitDirectory.isSymbolicLink(), "fixture .git must be a real directory, not a link or worktree pointer");
  const marker = JSON.parse(await read(root, markerPath));
  assert.equal(marker.kind, markerKind);
  assert.equal(marker.root, root);
  assert.equal(command(root, ["rev-parse", "--show-toplevel"]).trim(), root);
  assert.equal(command(root, ["rev-parse", "HEAD"]).trim(), marker.baselineHead, "fixture HEAD drift: do not restore");
  assert.equal(command(root, ["remote"]).trim(), "", "fixture must not have remotes");
  return marker;
}

const baselineMutablePaths = [
  indexPath, ...[0, 1, 2, 3, 4].map(reportPath),
  ...[1, 2, 3, 4].map((number) => resourcePath(formalId(number))),
  candidatePath(candidateA), candidatePath(candidateB), resourcePath(candidateA), "unrelated.txt",
];
const newFiles = [
  candidatePath(createdId), `${collection}/${nameOf(candidateA)}.md`,
  `${collection}/${nameOf(renamedId)}.md`, resourcePath(renamedId),
  resourcePath(formalId(2), "extra.txt"),
];

async function restore(root) {
  await guardedMarker(root);
  // 只恢复本模块会改动的已知tracked清单；不遍历清理其他未知文件。
  command(root, ["restore", "--source=HEAD", "--staged", "--worktree", "--", ...baselineMutablePaths]);
  command(root, ["rm", "--cached", "--ignore-unmatch", "--", ...newFiles]);
  for (const relative of newFiles) {
    const target = path.join(root, relative);
    try {
      const stat = await fs.lstat(target);
      assert.ok(stat.isFile() && !stat.isSymbolicLink(), `unexpected fixture artifact type: ${relative}`);
      await fs.unlink(target);
    } catch (error) { if (error?.code !== "ENOENT") throw error; }
  }
  for (const id of [renamedId, createdId]) {
    try { await fs.rmdir(path.join(root, `${collection}/_resources/${id}`)); }
    catch (error) { if (error?.code !== "ENOENT" && error?.code !== "ENOTEMPTY") throw error; }
  }
  assert.equal(command(root, ["diff", "--cached", "--name-only"]).trim(), "", "unknown pending survived fixture restore");
  assert.deepEqual(fullDeltaFiles(root), [], "unknown working-tree or untracked changes survived fixture restore");
}

async function patchSourcesIntoIndex(root, relatives) {
  const index = JSON.parse(await read(root, indexPath));
  for (const relative of relatives) {
    const text = await read(root, relative);
    const match = /^id: "([^"]+)"$/mu.exec(text);
    assert.ok(match);
    const source = { id: match[1], sourcePath: path.posix.basename(relative), text };
    const snapshot = buildInvestigationStateSnapshot([source]);
    index.entries[source.id] = snapshot.states[source.id];
    index.sourceRevision.entries[source.id] = investigationSourceRevision([source]).entries[source.id];
  }
  await write(root, indexPath, serializeStateIndex(index, definition));
}

function pendingFiles(root) {
  return command(root, ["diff", "--cached", "--name-only", "-z"]).split("\0").filter(Boolean).sort();
}

function pendingHash(root, relative) {
  return hash(command(root, ["show", `:${relative}`]));
}

function fullDeltaFiles(root) {
  // 不只检查操作清单：所有tracked差异及所有未跟踪文件（含ignored）都进入白名单核对。
  const tracked = command(root, ["diff", "--no-renames", "HEAD", "--name-only", "-z"]).split("\0").filter(Boolean);
  const untracked = command(root, ["ls-files", "--others", "-z"]).split("\0").filter(Boolean);
  return [...new Set([...tracked, ...untracked])].filter((relative) => relative !== markerPath && relative !== scenarioPath).sort();
}

export async function prepareScenario(root, id) {
  const selected = manifest.find((entry) => entry.id === id);
  assert.ok(selected, `unknown investigation scenario ${id}`);
  const marker = await guardedMarker(root);
  if (id === "search-content-cap") assert.ok(marker.count > 2000, "cap refusal requires N>2000; skip this scenario in N20/100 smoke");
  await restore(root);
  if (["sync-preview", "sync-write", "sync-selected"].includes(id)) {
    await write(root, reportPath(1), (await read(root, reportPath(1))).replace("The fixture is not", "One additional observation. The fixture is not"));
  }
  if (id.startsWith("stage-")) {
    const targetId = formalId(2);
    let selectedText = await read(root, reportPath(2));
    selectedText = selectedText.replace(`title: "Fixture ${nameOf(targetId)}"`, `title: "Fixture ${nameOf(targetId)} staged"`);
    selectedText += `- [new small evidence](./_resources/${targetId}/extra.txt)\n`;
    // 资源按ID词法排序：evidence.txt位于extra.txt之前。
    await write(root, reportPath(2), selectedText);
    await write(root, resourcePath(targetId), resourceText(targetId, true));
    await write(root, resourcePath(targetId, "extra.txt"), "new evidence\n".padEnd(95, "x") + "\n");
    await write(root, reportPath(4), (await read(root, reportPath(4))).replace(`title: "Fixture ${nameOf(formalId(4))}"`, `title: "Fixture ${nameOf(formalId(4))} unrelated"`));
    await write(root, resourcePath(formalId(4)), resourceText(formalId(4), true));
    await write(root, "unrelated.txt", "unrelated pending must be preserved\n");
    await patchSourcesIntoIndex(root, [reportPath(2), reportPath(4)]);
    command(root, ["add", "--", reportPath(4), resourcePath(formalId(4)), "unrelated.txt"]);
  }
  const before = {};
  for (const relative of [...baselineMutablePaths, ...newFiles]) before[relative] = await fileHash(root, relative);
  const state = {
    id, count: marker.count, before,
    baselineDelta: fullDeltaFiles(root),
    pendingTree: command(root, ["write-tree"]).trim(),
    protectedPending: Object.fromEntries(pendingFiles(root).map((relative) => [relative, pendingHash(root, relative)])),
  };
  await write(root, scenarioPath, JSON.stringify(state, null, 2) + "\n");
  return { id, count: marker.count, operationTargets: id.startsWith("stage-") ? [formalId(2)] : undefined, protectedPendingPaths: Object.keys(state.protectedPending) };
}

function assertIncludes(text, value) { assert.ok(text.includes(value), `expected output to include ${JSON.stringify(value)}; got ${text.slice(0, 1600)}`); }

export async function verifyScenario(root, id, result) {
  const marker = await guardedMarker(root);
  const selected = manifest.find((entry) => entry.id === id);
  assert.ok(selected);
  const state = JSON.parse(await read(root, scenarioPath));
  assert.equal(state.id, id);
  assert.equal(result.signal, null, "CLI must not be terminated by signal");
  assert.equal(result.exitCode, selected.expectedExit, `CLI ${id} failed: ${result.stderr}`);
  const stdout = result.stdout;
  const stderr = result.stderr;
  const indexText = await read(root, indexPath);
  const index = JSON.parse(indexText);
  const baselineIndex = JSON.parse(command(root, ["show", `HEAD:${indexPath}`]));
  const fullDelta = fullDeltaFiles(root);
  const changedFiles = [];
  for (const relative of new Set([...Object.keys(state.before), ...fullDelta])) {
    if ((state.before[relative] ?? null) !== await fileHash(root, relative)) changedFiles.push(relative);
  }
  const expectFiles = (expected) => {
    assert.deepEqual(changedFiles.sort(), [...expected].sort(), `unexpected working-tree changes for ${id}`);
    assert.deepEqual(fullDelta, [...new Set([...state.baselineDelta, ...expected])].sort(), `full HEAD/untracked delta escaped whitelist for ${id}`);
  };
  const pendingSame = () => assert.equal(command(root, ["write-tree"]).trim(), state.pendingTree, "non-stage command changed pending");
  const assertIndexDelta = (sources, removedIds = []) => {
    // 只从固定HEAD应用预期的单对象delta，再比较完整规范字节，保护全部未选entries及metadata。
    const expected = baselineIndex;
    for (const removedId of removedIds) {
      delete expected.entries[removedId];
      delete expected.sourceRevision.entries[removedId];
    }
    for (const source of sources) {
      const snapshot = buildInvestigationStateSnapshot([source]);
      expected.entries[source.id] = snapshot.states[source.id];
      expected.sourceRevision.entries[source.id] = snapshot.sourceRevision.entries[source.id];
    }
    assert.equal(hash(indexText), hash(serializeStateIndex(expected, definition)), `full index projection changed outside the expected delta for ${id}`);
  };
  if (id === "new") {
    expectFiles([candidatePath(createdId)]);
    assertIncludes(await read(root, candidatePath(createdId)), `id: "${createdId}"`);
    assertIncludes(stdout, createdId);
  } else if (id === "publish") {
    expectFiles([candidatePath(candidateA), `${collection}/${nameOf(candidateA)}.md`, indexPath]);
    assert.equal(index.entries[candidateA].resourceIds[0], `${candidateA}/evidence.txt`);
    assert.equal(Object.keys(index.entries).length, marker.count + 1);
    assert.equal(await fileHash(root, `${collection}/${nameOf(candidateA)}.md`), state.before[candidatePath(candidateA)]);
    assertIndexDelta([{ id: candidateA, sourcePath: `${nameOf(candidateA)}.md`, text: await read(root, `${collection}/${nameOf(candidateA)}.md`) }]);
    assertIncludes(stdout, "committed");
  } else if (id === "rename") {
    expectFiles([reportPath(1), resourcePath(formalId(1)), `${collection}/${nameOf(renamedId)}.md`, resourcePath(renamedId), indexPath]);
    assert.equal(index.entries[formalId(1)], undefined);
    assert.equal(index.entries[renamedId].resourceIds[0], `${renamedId}/evidence.txt`);
    assert.equal(await fileHash(root, resourcePath(renamedId)), state.before[resourcePath(formalId(1))]);
    assertIncludes(await read(root, `${collection}/${nameOf(renamedId)}.md`), `./_resources/${renamedId}/evidence.txt`);
    const expectedText = command(root, ["show", `HEAD:${reportPath(1)}`]).replaceAll(formalId(1), renamedId);
    assert.equal(await fileHash(root, `${collection}/${nameOf(renamedId)}.md`), hash(expectedText), "rename changed unrelated Markdown content");
    assertIndexDelta([{ id: renamedId, sourcePath: `${nameOf(renamedId)}.md`, text: expectedText }], [formalId(1)]);
  } else if (["sync-write", "sync-selected"].includes(id)) {
    expectFiles([indexPath]);
    const source = { id: formalId(1), sourcePath: path.posix.basename(reportPath(1)), text: await read(root, reportPath(1)) };
    assert.equal(index.sourceRevision.entries[source.id], investigationSourceRevision([source]).entries[source.id]);
    assertIndexDelta([source]);
  } else if (id === "set-relations") {
    expectFiles([reportPath(1), indexPath]);
    assert.equal(index.entries[formalId(1)].relations[0].summary, "本轮独立复核小组根节点");
    const expectedText = command(root, ["show", `HEAD:${reportPath(1)}`]).replace("在独立小样本中复核组根", "本轮独立复核小组根节点");
    assert.equal(await fileHash(root, reportPath(1)), hash(expectedText), "set-relations changed unrelated Markdown content");
    assertIndexDelta([{ id: formalId(1), sourcePath: path.posix.basename(reportPath(1)), text: expectedText }]);
    assertIncludes(stdout, "committed");
  } else if (id === "discard") {
    expectFiles([reportPath(3), resourcePath(formalId(3)), indexPath]);
    assert.equal(index.entries[formalId(3)], undefined);
    assert.equal(Object.keys(index.entries).length, marker.count - 1);
    assertIndexDelta([], [formalId(3)]);
  } else if (id.startsWith("stage-")) {
    expectFiles([]);
    const scope = id.slice(6);
    const domainPaths = [reportPath(2), resourcePath(formalId(2)), resourcePath(formalId(2), "extra.txt")];
    const expected = [...Object.keys(state.protectedPending), ...(scope === "index" ? [] : domainPaths), ...(scope === "domain" ? [] : [indexPath])].sort();
    assert.deepEqual(pendingFiles(root), expected, "stage scope wrote unexpected pending paths");
    for (const [relative, beforeHash] of Object.entries(state.protectedPending)) assert.equal(pendingHash(root, relative), beforeHash, `unselected pending changed: ${relative}`);
    if (scope !== "index") for (const relative of domainPaths) assert.equal(pendingHash(root, relative), await fileHash(root, relative), `selected pending is not current: ${relative}`);
    const expectedIndex = baselineIndex;
    if (scope !== "domain") {
      expectedIndex.entries[formalId(2)] = index.entries[formalId(2)];
      expectedIndex.sourceRevision.entries[formalId(2)] = index.sourceRevision.entries[formalId(2)];
    }
    assert.equal(pendingHash(root, indexPath), hash(serializeStateIndex(expectedIndex, definition)), "pending index did not preserve unselected published projection");
    assertIncludes(stdout, formalId(2));
  } else {
    expectFiles([]);
    if (id === "candidates") { assertIncludes(stdout, candidateA); assertIncludes(stdout, candidateB); }
    if (id === "show-candidate") { assertIncludes(stdout, candidateA); assertIncludes(stdout, "true"); }
    if (id === "check-scoped" || id === "check-full") assertIncludes(stdout, "check passed");
    if (id === "sync-preview") assertIncludes(stdout, "preflight");
    if (id === "sync-current") assertIncludes(stdout, "current");
    if (id === "list") assertIncludes(stdout, formalId(0));
    if (id === "search-metadata") { assertIncludes(stdout, "published-index"); assertIncludes(stdout, "matched=1 precision=exact returned=1"); assertIncludes(stdout, formalId(1)); }
    if (id === "search-content-cap") assert.match(stderr, /candidate file limit was exceeded/u);
    if (id === "search-content-related") { assertIncludes(stdout, "validated-source"); assertIncludes(stdout, "matched=9"); assertIncludes(stdout, "scan=complete results=complete previews=complete"); }
    if (id === "show") assertIncludes(stdout, `id: "${formalId(1)}"`);
    if (id === "trace") {
      const trace = JSON.parse(stdout);
      assert.equal(trace.anchorId, formalId(0));
      assert.equal(Object.keys(trace.entries).length, 10);
      assert.equal(trace.coverage.complete, true);
    }
  }
  if (!id.startsWith("stage-")) pendingSame();
  return { verified: true, id, changedFiles, pendingPaths: pendingFiles(root), formalCount: Object.keys(index.entries).length };
}

"""Measure one real CLI child with wait4; stdout/stderr remain inherited pipes."""
from __future__ import annotations

import json
import os
from pathlib import Path
import signal
import subprocess
import sys
import time


def main() -> int:
    if len(sys.argv) < 4 or sys.argv[2] != "--":
        raise SystemExit("usage: measure-process.py metrics.json -- executable [args...]")
    destination = Path(sys.argv[1])
    started_at = time.time()
    start = time.perf_counter_ns()
    child = subprocess.Popen(sys.argv[3:], stdin=subprocess.DEVNULL)
    while True:
        try:
            _, status, usage = os.wait4(child.pid, 0)
            break
        except InterruptedError:
            continue
    elapsed_ms = (time.perf_counter_ns() - start) / 1_000_000
    code = os.waitstatus_to_exitcode(status)
    child.returncode = code
    result = {
        "startedAtUnixSeconds": started_at,
        "wallMs": elapsed_ms,
        "userSeconds": usage.ru_utime,
        "systemSeconds": usage.ru_stime,
        "maxRssKiB": usage.ru_maxrss,
        "childExitCode": code if code >= 0 else None,
        "childSignal": signal.Signals(-code).name if code < 0 else None,
        "metricsScope": "os.wait4 CLI child rusage; RSS is not simultaneous process-tree memory",
    }
    temporary = destination.with_suffix(destination.suffix + ".tmp")
    temporary.write_text(json.dumps(result, ensure_ascii=False) + "\n", encoding="utf-8")
    temporary.replace(destination)
    return code if code >= 0 else 128 - code


if __name__ == "__main__":
    raise SystemExit(main())

import { createHash } from "node:crypto";
import fs from "node:fs";
import { mkdir, readFile } from "node:fs/promises";
import path from "node:path";
import { spawn } from "node:child_process";
import { finished } from "node:stream/promises";

const captureLimit = 256 * 1024;

function outputSink(file) {
  const stream = fs.createWriteStream(file, { flags: "wx" });
  const hash = createHash("sha256");
  const captured = [];
  let bytes = 0;
  let capturedBytes = 0;
  return {
    push(chunk) {
      hash.update(chunk);
      bytes += chunk.length;
      if (capturedBytes < captureLimit) {
        const part = chunk.subarray(0, captureLimit - capturedBytes);
        captured.push(part);
        capturedBytes += part.length;
      }
      return stream.write(chunk);
    },
    drain: () => new Promise((resolve) => stream.once("drain", resolve)),
    async close() {
      stream.end();
      await finished(stream);
      return {
        bytes,
        sha256: hash.digest("hex"),
        head: Buffer.concat(captured).toString("utf8"),
        headTruncated: bytes > capturedBytes,
        file
      };
    }
  };
}

function signalGroup(child, signal) {
  if (child.pid === undefined) return;
  try {
    process.kill(-child.pid, signal);
  } catch (error) {
    if (error.code !== "ESRCH") throw error;
  }
}

export async function runCaptured({ command, args, cwd, timeoutMs, prefix }) {
  await mkdir(path.dirname(prefix), { recursive: true });
  const stdout = outputSink(prefix + ".stdout");
  const stderr = outputSink(prefix + ".stderr");
  const startedAt = new Date().toISOString();
  const start = performance.now();
  const child = spawn(command, args, {
    cwd,
    detached: true,
    stdio: ["ignore", "pipe", "pipe"],
    env: { ...process.env }
  });
  let timedOut = false;
  let spawnError = null;
  let forced = null;
  const timer = setTimeout(() => {
    timedOut = true;
    signalGroup(child, "SIGTERM");
    forced = setTimeout(() => signalGroup(child, "SIGKILL"), 5000);
  }, timeoutMs);
  const consume = async (source, sink) => {
    for await (const chunk of source) {
      if (!sink.push(chunk)) await sink.drain();
    }
  };
  const streams = Promise.all([
    consume(child.stdout, stdout),
    consume(child.stderr, stderr)
  ]);
  const status = await new Promise((resolve) => {
    child.once("error", (error) => {
      spawnError = { code: error.code ?? null, message: error.message };
    });
    child.once("close", (exitCode, signal) => resolve({ exitCode, signal }));
  });
  clearTimeout(timer);
  if (forced !== null) clearTimeout(forced);
  if (timedOut) signalGroup(child, "SIGKILL");
  await streams;
  return {
    startedAt,
    wrapperWallMs: performance.now() - start,
    ...status,
    timedOut,
    spawnError,
    stdout: await stdout.close(),
    stderr: await stderr.close()
  };
}

export async function readCgroupMemoryEvents() {
  try {
    const entries = (await readFile("/sys/fs/cgroup/memory.events", "utf8"))
      .trim()
      .split("\n")
      .map((line) => {
        const [key, raw] = line.split(/\s+/u);
        const value = Number(raw);
        if (!key || !Number.isSafeInteger(value) || value < 0)
          throw new Error("invalid memory.events");
        return [key, value];
      });
    return Object.fromEntries(entries);
  } catch (error) {
    return { unavailable: error.code ?? error.message };
  }
}
